cmake_minimum_required(VERSION 3.20)

project(
  morphos
  VERSION 0.1.0
  LANGUAGES CXX
)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

add_library(morphos_core
  src/cell.cpp
  src/genome.cpp
  src/world.cpp
  src/simulation.cpp
  src/main.cpp
)

target_include_directories(
  morphos_core
  PUBLIC
  ${PROJECT_SOURCE_DIR}/include
)

add_executable(
  morphos
  src/main.cpp
)

target_link_libraries(
  morphos
  PRIVATE
  morphos_core
)
