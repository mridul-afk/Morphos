
cmake_minimum_required(VERSION 3.20)

project(
  morphos
  VERSION 0.1.0
  LANGUAGES CXX
)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

# Core simulation engine
add_library(morphos_core
  src/cell.cpp
  src/genome.cpp
  src/world.cpp
  src/simulation.cpp
)

target_include_directories(morphos_core
  PUBLIC
  ${PROJECT_SOURCE_DIR}/include
)

# Main executable
add_executable(morphos src/main.cpp)

target_link_libraries(morphos
  PRIVATE morphos_core
)

# GoogleTest
include(CTest)

if(BUILD_TESTING)
  include(FetchContent)

  FetchContent_Declare(
    googletest
    URL https://github.com/google/googletest/archive/03597a01ee50ed33e9dfd640b249b4be3799d395.zip
    DOWNLOAD_EXTRACT_TIMESTAMP TRUE
  )

  # Match MSVC runtime settings
  set(gtest_force_shared_crt ON CACHE BOOL "" FORCE)
  set(BUILD_GMOCK OFF CACHE BOOL "" FORCE)

  FetchContent_MakeAvailable(googletest)

  add_executable(morphos_tests
    tests/test_world.cpp
    tests/test_cell.cpp
    tests/test_simulation.cpp
  )

  target_link_libraries(morphos_tests
    PRIVATE
    morphos_core
    GTest::gtest_main
  )

  include(GoogleTest)
  gtest_discover_tests(morphos_tests)
endif()
